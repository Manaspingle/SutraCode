def fetch_user_data(user_id):
    url = f"https://api.example.com/users/{user_id}"
    # ✅ SUTRATRACE FIX: Connect (3.05s) / Read (10s) timeout
    response = requests.get(url, timeout=(3.05, 10))
    response.raise_for_status()
    return response.json()
